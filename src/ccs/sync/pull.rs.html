36fn write_session_within_local_root(
37    session: &ConversationSession,
38    local_root: &Path,
39    relative: &Path,
40) -> Result<PathBuf> {
41    prepare_local_session_destination(local_root, relative)?;
42    // Rebuild and revalidate immediately before the write. A same-UID
43    // replacement after this check remains the documented narrow TOCTOU residual.
44    let destination = prepare_local_session_destination(local_root, relative)?;
45    session.write_to_file(&destination)?;
46    Ok(destination)
47}

49fn propagate_tombstones(local_projects_root: &Path, registry: &TombstoneRegistry) -> Result<usize> {
50    validate_directory_root(local_projects_root)?;
51    let mut propagated = 0;
52    let entries = match fs::read_dir(local_projects_root) {
53        Ok(entries) => entries,
54        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(0),
55        Err(error) => return Err(error.into()),
56    };
57
58    for entry in entries {
59        let entry = entry?;
60        let local_project_dir = entry.path();
61        let metadata = fs::symlink_metadata(&local_project_dir)?;
62        if metadata.file_type().is_symlink() || !metadata.is_dir() {
63            log::warn!(
64                "Skipping unsafe local project while applying tombstones: {}",
65                local_project_dir.display()
66            );
67            continue;
68        }
69        if let Err(error) = validate_directory_candidate(local_projects_root, &local_project_dir) {
70            log::warn!(
71                "Skipping unsafe local project while applying tombstones: {}",
72                error
73            );
74            continue;
75        }
76
77        for file in fs::read_dir(&local_project_dir)? {
78            let file = file?;
79            let file_name = file.file_name();
80            let Some(name) = file_name.to_str() else {
81                continue;
82            };
83            if !name.ends_with(".jsonl") {
84                continue;
85            }
86
87            let session_id = name
88                .strip_suffix(".jsonl")
89                .unwrap_or(name)
90                .trim_start_matches("session-");
91            if !registry.contains(session_id) {
92                continue;
93            }
94
95            let file_path = file.path();
96            let metadata = fs::symlink_metadata(&file_path)?;
97            if metadata.file_type().is_symlink() || !metadata.is_file() {
98                log::warn!(
99                    "Skipping unsafe tombstone deletion candidate: {}",
100                    file_path.display()
101                );
102                continue;
103            }
104            validate_regular_candidate(local_projects_root, &file_path)?;
105            let relative = file_path
106                .strip_prefix(local_projects_root)
107                .with_context(|| {
108                    format!(
109                        "local session path is outside projects root: {}",
110                        file_path.display()
111                    )
112                })?;
113            let candidate = safe_join_within_root(local_projects_root, relative)?;
114            validate_regular_candidate(local_projects_root, &candidate)?;
115            fs::remove_file(&candidate)
116                .with_context(|| format!("failed to propagate tombstone for {session_id}"))?;
117            propagated += 1;
118            log::debug!("Propagated remote deletion: {}", candidate.display());
119        }
120    }
121
122    Ok(propagated)
123}

125fn sync_auto_memory_from_remote(
126    sync_repo_path: &Path,
127    remote_projects_dir: &Path,
128    local_projects_root: &Path,
129    use_project_name_only: bool,
130) -> Result<Vec<String>> {
131    validate_sync_projects_root(sync_repo_path, remote_projects_dir)?;
132    validate_directory_root(local_projects_root)?;
133    let mut synced_projects = Vec::new();
134
135    for entry in fs::read_dir(remote_projects_dir)? {
136        let entry = entry?;
137        let project_name = match entry.file_name().to_str() {
138            Some(name) if !name.starts_with('.') && !name.is_empty() => name.to_string(),
139            _ => continue,
140        };
141        validate_project_component(&project_name)?;
142
143        let project_relative = PathBuf::from(&project_name);
144        let sync_project_dir = safe_join_within_sync_projects_root(
145            sync_repo_path,
146            remote_projects_dir,
147            &project_relative,
148        )?;
149        let project_metadata = fs::symlink_metadata(&sync_project_dir)?;
150        if project_metadata.file_type().is_symlink() {
151            anyhow::bail!("remote auto-memory project path must not be a symlink");
152        }
153        if !project_metadata.is_dir() {
154            continue;
155        }
156        validate_directory_candidate(remote_projects_dir, &sync_project_dir)?;
157
158        let memory_relative = project_relative.join("memory");
159        let remote_memory_path = safe_join_within_sync_projects_root(
160            sync_repo_path,
161            remote_projects_dir,
162            &memory_relative,
163        )?;
164        let remote_memory_metadata = match fs::symlink_metadata(&remote_memory_path) {
165            Ok(metadata) => metadata,
166            Err(error) if error.kind() == std::io::ErrorKind::NotFound => continue,
167            Err(error) => return Err(error.into()),
168        };
169        if remote_memory_metadata.file_type().is_symlink() || !remote_memory_metadata.is_dir() {
170            anyhow::bail!("remote auto-memory path must be a non-symlink directory");
171        }
172        validate_directory_candidate(remote_projects_dir, &remote_memory_path)?;
173
174        let local_project_dir = if use_project_name_only {
175            find_local_project_by_name(local_projects_root, &project_name)
176        } else {
177            let local_path = safe_join_within_root(local_projects_root, &project_relative)?;
178            match fs::symlink_metadata(&local_path) {
179                Ok(metadata) if metadata.is_dir() && !metadata.file_type().is_symlink() => {
180                    Some(local_path)
181                }
182                Ok(_) => None,
183                Err(error) if error.kind() == std::io::ErrorKind::NotFound => None,
184                Err(error) => return Err(error.into()),
185            }
186        };
187
188        let Some(local_project_dir) = local_project_dir else {
189            log::debug!(
190                "No local project found for '{}', skipping memory sync",
191                project_name
192            );
193            continue;
194        };
195        validate_directory_candidate(local_projects_root, &local_project_dir)?;
196
197        let local_memory_path = safe_join_within_root(&local_project_dir, Path::new("memory"))?;
198        match fs::symlink_metadata(&local_memory_path) {
199            Ok(metadata) if metadata.file_type().is_symlink() || !metadata.is_dir() => {
200                anyhow::bail!("local auto-memory path must be a non-symlink directory");
201            }
202            Ok(_) => {}
203            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
204                fs::create_dir_all(&local_memory_path)?;
205            }
206            Err(error) => return Err(error.into()),
207        }
208        validate_directory_candidate(local_projects_root, &local_memory_path)?;
209
210        for memory_entry in fs::read_dir(&remote_memory_path)? {
211            let memory_entry = memory_entry?;
212            let source = memory_entry.path();
213            let source_metadata = fs::symlink_metadata(&source)?;
214            if source_metadata.file_type().is_symlink() {
215                anyhow::bail!("remote auto-memory contains a symlink file");
216            }
217            if !source_metadata.is_file() {
218                continue;
219            }
220            validate_regular_candidate(remote_projects_dir, &source)?;
221
222            let destination =
223                safe_join_within_root(&local_memory_path, Path::new(&memory_entry.file_name()))?;
224            if let Ok(metadata) = fs::symlink_metadata(&destination) {
225                if metadata.file_type().is_symlink() || !metadata.is_file() {
226                    anyhow::bail!("local auto-memory destination is not a regular file");
227                }
228            }
229
230            validate_regular_candidate(remote_projects_dir, &source)?;
231            let destination =
232                safe_join_within_root(&local_memory_path, Path::new(&memory_entry.file_name()))?;
233            fs::copy(&source, &destination)?;
234        }
235        synced_projects.push(project_name);
236    }
237
238    Ok(synced_projects)
239}

241/// Pull and merge history from sync repository
242pub fn pull_history(
243    fetch_remote: bool,
244    branch: Option<&str>,
245    interactive: bool,
246    verbosity: crate::VerbosityLevel,
247) -> Result<()> {
248    use crate::VerbosityLevel;
249
250    if verbosity != VerbosityLevel::Quiet {
251        println!("{}", "Pulling Claude Code history...".cyan().bold());
252    }
253
254    let state = SyncState::load()?;
255    let repo = scm::open(&state.sync_repo_path)?;
256    let filter = FilterConfig::load()?;
257    let claude_dir = claude_projects_dir()?;
258    validate_directory_root(&claude_dir)?;
259
260    // Get the current branch name for operation record
261    let branch_name = branch
262        .map(|s| s.to_string())
263        .or_else(|| repo.current_branch().ok())
264        .unwrap_or_else(|| "main".to_string());
265
266    // Fetch from remote if configured
267    if fetch_remote && state.has_remote {
268        println!("  {} from remote...", "Fetching".cyan());
269
270        match repo.pull("origin", &branch_name) {
271            Ok(_) => println!("  {} Pulled from origin/{}", "✓".green(), branch_name),
272            Err(e) => {
273                log::warn!("Failed to pull: {}", e);
274                log::info!("Continuing with local sync repository state...");
275            }
276        }
277    }
278
279    // ============================================================================
280    // PROPAGATE INTENTIONAL DELETIONS
281    // ============================================================================
282    // Before discovering local sessions, check if the sync repo has any
283    // registered tombstones that we haven't applied locally yet.
284    match TombstoneRegistry::load(&state.sync_repo_path) {
285        Ok(registry) if !registry.is_empty() => {
286            if verbosity != VerbosityLevel::Quiet {
287                println!("  {} tombstones...", "Checking".cyan());
288            }
289            let propagated_deletes = propagate_tombstones(&claude_dir, &registry)?;
290            if propagated_deletes > 0 && verbosity != VerbosityLevel::Quiet {
291                println!(
292                    "  {} Propagated {} intentional deletion(s) from other devices",
293                    "✓".green(),
294                    propagated_deletes
295                );
296            }
297        }
298        Ok(_) => {}
299        Err(error) => {
300            log::warn!("Failed to load tombstone registry safely: {}", error);
301        }
302    }
303
304    // Discover local sessions
305    println!("  {} local sessions...", "Discovering".cyan());
306    let local_sessions = discover_sessions(&claude_dir, &filter)?;
307    println!(
308        "  {} {} local sessions",
309        "Found".green(),
310        local_sessions.len()
311    );
312
313    // Discover remote sessions
314    let remote_projects_dir = state.sync_repo_path.join(&filter.sync_subdirectory);
315    validate_sync_projects_root(&state.sync_repo_path, &remote_projects_dir)?;
316    println!("  {} remote sessions...", "Discovering".cyan());
317    let remote_sessions = discover_sessions(&remote_projects_dir, &filter)?;
318    println!(
319        "  {} {} remote sessions",
320        "Found".green(),
321        remote_sessions.len()
322    );
323
324    // ============================================================================
325    // CONFLICT DETECTION (moved before snapshot for efficiency)
326    // ============================================================================
327    // Detect conflicts FIRST so we only backup files that will be modified
328    if verbosity != VerbosityLevel::Quiet {
329        println!("  {} conflicts...", "Detecting".cyan());
330    }
331    let mut detector = ConflictDetector::new();
332    detector.detect(&local_sessions, &remote_sessions);
333
334    // ============================================================================
335    // SNAPSHOT CREATION: Only backup files that have conflicts
336    // ============================================================================
337    // Optimization: Only backup local files that have conflicts and will be merged.
338    // Files that are new (remote-only) or unchanged don't need backup.
339    // This reduces snapshot size from potentially gigabytes to typically <1MB.
340    let snapshot_path = if detector.has_conflicts() {
341        println!(
342            "  {} snapshot of {} conflicting files...",
343            "Creating".cyan(),
344            detector.conflict_count()
345        );
346
347        // Only collect paths for files that have conflicts
348        let conflicting_file_paths: Vec<PathBuf> = detector
349            .conflicts()
350            .iter()
351            .map(|c| c.local_file.clone())
352            .collect();
353
354        // Check for large conversation files and warn users
355        warn_large_files(&conflicting_file_paths);
356
357        // Create snapshot of ONLY conflicting files
358        let snapshot = Snapshot::create(
359            OperationType::Pull,
360            conflicting_file_paths.iter(),
361            None, // No git manager needed for pull snapshots
362        )
363        .context("Failed to create snapshot before pull")?;
364
365        // Save snapshot to disk
366        let path = snapshot
367            .save_to_disk(None)
368            .context("Failed to save snapshot to disk")?;
369
370        if verbosity != VerbosityLevel::Quiet {
371            println!(
372                "  {} Snapshot created: {} ({} files)",
373                "✓".green(),
374                path.file_name()
375                    .map(|n| n.to_string_lossy().to_string())
376                    .unwrap_or_else(|| path.display().to_string()),
377                conflicting_file_paths.len()
378            );
379        }
380
381        Some(path)
382    } else {
383        println!("  {} No conflicts - skipping snapshot", "✓".green());
384        None
385    };
386
387    // ============================================================================
388    // SHOW SUMMARY AND INTERACTIVE CONFIRMATION
389    // ============================================================================
390    if verbosity != VerbosityLevel::Quiet {
391        println!();
392        println!("{}", "Pull Summary:".bold().cyan());
393        println!("  {} Local sessions: {}", "•".cyan(), local_sessions.len());
394        println!(
395            "  {} Remote sessions: {}",
396            "•".cyan(),
397            remote_sessions.len()
398        );
399        println!();
400    }
401
402    // Show detailed file list in verbose mode
403    if verbosity == VerbosityLevel::Verbose {
404        println!("{}", "Remote sessions to be pulled:".bold());
405        for (idx, session) in remote_sessions.iter().enumerate().take(20) {
406            let relative_path = Path::new(&session.file_path)
407                .strip_prefix(&remote_projects_dir)
408                .unwrap_or(Path::new(&session.file_path));
409
410            println!(
411                "  {}. {} ({} messages)",
412                idx + 1,
413                relative_path.display(),
414                session.message_count()
415            );
416        }
417        if remote_sessions.len() > 20 {
418            println!("  ... and {} more", remote_sessions.len() - 20);
419        }
420        println!();
421    }
422
423    // Interactive confirmation
424    if interactive && interactive_conflict::is_interactive() {
425        let confirm =
426            Confirm::new("Do you want to proceed with pulling and merging these changes?")
427                .with_default(true)
428                .with_help_message(
429                    "This will merge remote sessions into your local Claude Code history",
430                )
431                .prompt()
432                .context("Failed to get confirmation")?;
433
434        if !confirm {
435            println!("\n{}", "Pull cancelled.".yellow());
436            return Ok(());
437        }
438    }
439
440    // ============================================================================
441    // CONFLICT RESOLUTION (detection already done above)
442    // ============================================================================
443    // Track affected conversations for operation record
444    let mut affected_conversations: Vec<ConversationSummary> = Vec::new();
445
446    if detector.has_conflicts() {
447        println!(
448            "  {} {} conflicts detected",
449            "!".yellow(),
450            detector.conflict_count()
451        );
452
453        // ============================================================================
454        // ATTEMPT SMART MERGE FIRST
455        // ============================================================================
456        println!("  {} smart merge...", "Attempting".cyan());
457
458        let local_map: HashMap<_, _> = local_sessions
459            .iter()
460            .map(|s| (s.session_id.clone(), s))
461            .collect();
462
463        let remote_map: HashMap<_, _> = remote_sessions
464            .iter()
465            .map(|s| (s.session_id.clone(), s))
466            .collect();
467
468        let mut smart_merge_success_count = 0;
469        let mut smart_merge_failed_conflicts = Vec::new();
470
471        for conflict in detector.conflicts_mut() {
472            // Find local and remote sessions
473            if let (Some(local_session), Some(remote_session)) = (
474                local_map.get(&conflict.session_id),
475                remote_map.get(&conflict.session_id),
476            ) {
477                // Try smart merge
478                match conflict.try_smart_merge(local_session, remote_session) {
479                    Ok(()) => {
480                        smart_merge_success_count += 1;
481                        // Write merged result to local file
482                        if let crate::conflict::ConflictResolution::SmartMerge {
483                            ref merged_entries,
484                            ref stats,
485                        } = conflict.resolution
486                        {
487                            // Create a new session with merged entries
488                            let merged_session = ConversationSession {
489                                session_id: conflict.session_id.clone(),
490                                entries: merged_entries.clone(),
491                                file_path: conflict.local_file.to_string_lossy().to_string(),
492                            };
493
494                            // Rebuild the destination from the trusted local root.
495                            let local_relative = conflict
496                                .local_file
497                                .strip_prefix(&claude_dir)
498                                .context("conflict destination is outside Claude projects root")?;
499                            if let Err(e) = write_session_within_local_root(
500                                &merged_session,
501                                &claude_dir,
502                                local_relative,
503                            ) {
504                                log::warn!(
505                                    "Failed to write merged session {}: {}",
506                                    conflict.session_id,
507                                    e
508                                );
509                                smart_merge_failed_conflicts.push(conflict.clone());
510                            } else {
511                                println!(
512                                    "  {} Smart merged {} ({} local + {} remote = {} total, {} branches)",
513                                    "✓".green(),
514                                    conflict.session_id,
515                                    stats.local_messages,
516                                    stats.remote_messages,
517                                    stats.merged_messages,
518                                    stats.branches_detected
519                                );
520                            }
521                        }
522                    }
523                    Err(e) => {
524                        log::warn!("Smart merge failed for {}: {}", conflict.session_id, e);
525                        log::info!("Falling back to manual resolution...");
526                        smart_merge_failed_conflicts.push(conflict.clone());
527                    }
528                }
529            }
530        }
531
532        println!(
533            "  {} Successfully smart merged {}/{} conflicts",
534            "✓".green(),
535            smart_merge_success_count,
536            detector.conflict_count()
537        );
538
539        // If some smart merges failed, handle them with interactive/keep-both resolution
540        let renames = if !smart_merge_failed_conflicts.is_empty() {
541            println!(
542                "  {} {} conflicts require manual resolution",
543                "!".yellow(),
544                smart_merge_failed_conflicts.len()
545            );
546
547            // Check if we can run interactively
548            let use_interactive = crate::interactive_conflict::is_interactive();
549
550            if use_interactive {
551                // Interactive conflict resolution for failed merges
552                println!(
553                    "\n{} Running in interactive mode for remaining conflicts",
554                    "→".cyan()
555                );
556
557                let resolution_result = crate::interactive_conflict::resolve_conflicts_interactive(
558                    &mut smart_merge_failed_conflicts,
559                )?;
560
561                // Apply the resolutions
562                let renames = crate::interactive_conflict::apply_resolutions(
563                    &resolution_result,
564                    &remote_sessions,
565                    &claude_dir,
566                    &remote_projects_dir,
567                )?;
568
569                // Save conflict report
570                let report = ConflictReport::from_conflicts(detector.conflicts());
571                save_conflict_report(&report)?;
572
573                renames
574            } else {
575                // Non-interactive mode: use "keep both" strategy for failed merges
576                println!(
577                    "\n{} Using automatic conflict resolution (keep both versions)",
578                    "→".cyan()
579                );
580
581                let mut renames = Vec::new();
582
583                println!("\n{}", "Conflict Resolution:".yellow().bold());
584                for conflict in &smart_merge_failed_conflicts {
585                    let timestamp = chrono::Utc::now().format("%Y%m%d-%H%M%S");
586                    let conflict_suffix = format!("conflict-{timestamp}");
587
588                    if let Ok(renamed_path) = conflict.clone().resolve_keep_both(&conflict_suffix) {
589                        let relative_renamed = renamed_path
590                            .strip_prefix(&claude_dir)
591                            .unwrap_or(&renamed_path);
592                        println!(
593                            "  {} remote version saved as: {}",
594                            "→".yellow(),
595                            relative_renamed.display().to_string().cyan()
596                        );
597
598                        // Find and write the remote session through the trusted local root.
599                        if let Some(session) = remote_sessions
600                            .iter()
601                            .find(|s| s.session_id == conflict.session_id)
602                        {
603                            let renamed_relative = renamed_path.strip_prefix(&claude_dir).context(
604                                "conflict copy destination is outside Claude projects root",
605                            )?;
606                            write_session_within_local_root(
607                                session,
608                                &claude_dir,
609                                renamed_relative,
610                            )?;
611                        }
612
613                        renames.push((conflict.remote_file.clone(), renamed_path));
614                    }
615                }
616
617                // Save conflict report
618                let report = ConflictReport::from_conflicts(detector.conflicts());
619                save_conflict_report(&report)?;
620
621                renames
622            }
623        } else {
624            // All conflicts resolved via smart merge
625            Vec::new()
626        };
627
628        // Track all conflicts in affected conversations
629        for (_original_path, renamed_path) in &renames {
630            let relative_path = renamed_path
631                .strip_prefix(&claude_dir)
632                .unwrap_or(renamed_path)
633                .to_string_lossy()
634                .to_string();
635
636            // Find the session ID from the renamed path
637            if let Some(session) = remote_sessions.iter().find(|s| {
638                let session_file = Path::new(&s.file_path).file_name();
639                let renamed_file = renamed_path.file_name();
640                // Try to match based on session ID in filename
641                session_file
642                    .and_then(|f| f.to_str())
643                    .and_then(|name| name.split('-').next())
644                    == renamed_file
645                        .and_then(|f| f.to_str())
646                        .and_then(|name| name.split('-').next())
647            }) {
648                match ConversationSummary::new(
649                    session.session_id.clone(),
650                    relative_path.clone(),
651                    session.latest_timestamp(),
652                    session.message_count(),
653                    SyncOperation::Conflict,
654                ) {
655                    Ok(summary) => affected_conversations.push(summary),
656                    Err(e) => log::warn!(
657                        "Failed to create summary for conflict {}: {}",
658                        relative_path,
659                        e
660                    ),
661                }
662            }
663        }
664
665        println!(
666            "\n{} View details with: {} report",
667            "Hint:".cyan(),
668            BINARY_NAME
669        );
670    } else {
671        println!("  {} No conflicts detected", "✓".green());
672    }
673
674    // ============================================================================
675    // MERGE NON-CONFLICTING SESSIONS
676    // ============================================================================
677    println!("  {} non-conflicting sessions...", "Merging".cyan());
678    let local_map: HashMap<_, _> = local_sessions
679        .iter()
680        .map(|s| (s.session_id.clone(), s))
681        .collect();
682
683    let mut merged_count = 0;
684    let mut added_count = 0;
685    let mut modified_count = 0;
686    let mut unchanged_count = 0;
687    let mut skipped_no_local_match = 0;
688
689    for remote_session in &remote_sessions {
690        // Skip if conflicts were detected
691        if detector
692            .conflicts()
693            .iter()
694            .any(|c| c.session_id == remote_session.session_id)
695        {
696            continue;
697        }
698
699        let relative_path_for_tracking = if filter.use_project_name_only {
700            // Extract project name and session filename from remote path.
701            let remote_relative = Path::new(&remote_session.file_path)
702                .strip_prefix(&remote_projects_dir)
703                .ok()
704                .unwrap_or_else(|| Path::new(&remote_session.file_path));
705            let project_name = remote_relative
706                .components()
707                .next()
708                .and_then(|component| component.as_os_str().to_str())
709                .unwrap_or("unknown");
710            validate_project_component(project_name)?;
711
712            let Some(local_project_dir) = find_local_project_by_name(&claude_dir, project_name)
713            else {
714                log::debug!(
715                    "No matching local project found for '{}', skipping",
716                    project_name
717                );
718                skipped_no_local_match += 1;
719                continue;
720            };
721            validate_directory_candidate(&claude_dir, &local_project_dir)?;
722
723            let Some(filename) = remote_relative.file_name() else {
724                log::warn!(
725                    "Could not extract filename from remote path: {:?}",
726                    remote_relative
727                );
728                skipped_no_local_match += 1;
729                continue;
730            };
731            let local_project_relative = local_project_dir
732                .strip_prefix(&claude_dir)
733                .context("matched local project is outside Claude projects root")?;
734            local_project_relative.join(filename)
735        } else {
736            Path::new(&remote_session.file_path)
737                .strip_prefix(&remote_projects_dir)
738                .context("remote session is outside the sync projects root")?
739                .to_path_buf()
740        };
741
742        prepare_local_session_destination(&claude_dir, &relative_path_for_tracking)?;
743
744        // Determine operation type based on local state
745        let operation = if let Some(local) = local_map.get(&remote_session.session_id) {
746            if local.content_hash() == remote_session.content_hash() {
747                unchanged_count += 1;
748                SyncOperation::Unchanged
749            } else {
750                modified_count += 1;
751                SyncOperation::Modified
752            }
753        } else {
754            added_count += 1;
755            SyncOperation::Added
756        };
757
758        // Copy file if it's not unchanged
759        if operation != SyncOperation::Unchanged {
760            write_session_within_local_root(
761                remote_session,
762                &claude_dir,
763                &relative_path_for_tracking,
764            )?;
765            merged_count += 1;
766        }
767
768        // Track all sessions (including unchanged) in affected conversations
769        let relative_path_str = relative_path_for_tracking.to_string_lossy().to_string();
770        match ConversationSummary::new(
771            remote_session.session_id.clone(),
772            relative_path_str.clone(),
773            remote_session.latest_timestamp(),
774            remote_session.message_count(),
775            operation,
776        ) {
777            Ok(summary) => affected_conversations.push(summary),
778            Err(e) => log::warn!("Failed to create summary for {}: {}", relative_path_str, e),
779        }
780    }
781
782    println!("  {} Merged {} sessions", "✓".green(), merged_count);
783
784    // ============================================================================
785    // CREATE AND SAVE OPERATION RECORD
786    // ============================================================================
787    let mut operation_record = OperationRecord::new(
788        OperationType::Pull,
789        Some(branch_name.clone()),
790        affected_conversations.clone(),
791    );
792
793    // Attach the snapshot path to the operation record (only if we created one)
794    operation_record.snapshot_path = snapshot_path;
795
796    // Load operation history and add this operation
797    let mut history = match OperationHistory::load() {
798        Ok(h) => h,
799        Err(e) => {
800            log::warn!("Failed to load operation history: {}", e);
801            log::info!("Creating new history...");
802            OperationHistory::default()
803        }
804    };
805
806    if let Err(e) = history.add_operation(operation_record) {
807        log::warn!("Failed to save operation to history: {}", e);
808        log::info!("Pull completed successfully, but history was not updated.");
809    }
810
811    // ============================================================================
812    // DISPLAY SUMMARY TO USER
813    // ============================================================================
814    println!("\n{}", "=== Pull Summary ===".bold().cyan());
815
816    // Show operation statistics
817    let conflict_count = detector.conflict_count();
818    let stats_msg = format!(
819        "  {} Added    {} Modified    {} Conflicts    {} Unchanged",
820        format!("{added_count}").green(),
821        format!("{modified_count}").cyan(),
822        format!("{conflict_count}").yellow(),
823        format!("{unchanged_count}").dimmed(),
824    );
825    println!("{stats_msg}");
826    if filter.use_project_name_only && skipped_no_local_match > 0 {
827        println!(
828            "  {} Skipped (no local match): {}",
829            "!".yellow(),
830            skipped_no_local_match
831        );
832    }
833    println!();
834
835    // Group conversations by project (top-level directory)
836    let mut by_project: HashMap<String, Vec<&ConversationSummary>> = HashMap::new();
837    for conv in &affected_conversations {
838        // Skip unchanged conversations in detailed output
839        if conv.operation == SyncOperation::Unchanged {
840            continue;
841        }
842
843        let project = conv
844            .project_path
845            .split('/')
846            .next()
847            .unwrap_or("unknown")
848            .to_string();
849        by_project.entry(project).or_default().push(conv);
850    }
851
852    // Display conversations grouped by project
853    if !by_project.is_empty() {
854        println!("{}", "Affected Conversations:".bold());
855
856        let mut projects: Vec<_> = by_project.keys().collect();
857        projects.sort();
858
859        for project in projects {
860            let conversations = &by_project[project];
861            println!("\n  {} {}/", "Project:".bold(), project.cyan());
862
863            for conv in conversations.iter().take(MAX_CONVERSATIONS_TO_DISPLAY) {
864                let operation_str = match conv.operation {
865                    SyncOperation::Added => "ADD".green(),
866                    SyncOperation::Modified => "MOD".cyan(),
867                    SyncOperation::Conflict => "CONFLICT".yellow(),
868                    SyncOperation::Unchanged => "---".dimmed(),
869                };
870
871                let timestamp_str = conv
872                    .timestamp
873                    .as_ref()
874                    .and_then(|t| {
875                        // Extract just the date portion for compact display
876                        t.split('T').next()
877                    })
878                    .unwrap_or("unknown");
879
880                println!(
881                    "    {} {} ({}msg, {})",
882                    operation_str,
883                    conv.project_path,
884                    conv.message_count,
885                    timestamp_str.dimmed()
886                );
887            }
888
889            if conversations.len() > MAX_CONVERSATIONS_TO_DISPLAY {
890                println!(
891                    "    {} ... and {} more conversations",
892                    "...".dimmed(),
893                    conversations.len() - MAX_CONVERSATIONS_TO_DISPLAY
894                );
895            }
896        }
897    }
898
899    println!("\n{}", "Pull complete!".green().bold());
900
901    // Clean up old snapshots automatically
902    if let Err(e) = crate::undo::cleanup_old_snapshots(None, false) {
903        log::warn!("Failed to cleanup old snapshots: {}", e);
904    }
905
906    // ============================================================================
907    // SYNC AUTO MEMORY DIRECTORIES
908    // ============================================================================
909    if filter.auto_memory.enabled {
910        println!("  {} auto memory directories...", "Syncing".cyan());
911
912        let synced_projects = sync_auto_memory_from_remote(
913            &state.sync_repo_path,
914            &remote_projects_dir,
915            &claude_dir,
916            filter.use_project_name_only,
917        )?;
918        if verbosity == VerbosityLevel::Verbose {
919            for project_name in &synced_projects {
920                println!("    {} {}/memory", "←".cyan(), project_name);
921            }
922        }
923        let synced_count = synced_projects.len();
924
925        if verbosity != VerbosityLevel::Quiet {
926            println!(
927                "  {} Synced {} memory directories",
928                "✓".green(),
929                synced_count
930            );
931        }
932    }
933
934    // Auto-apply CLAUDE.md if enabled
935    if filter.config_sync.enabled && filter.config_sync.auto_apply_claude_md {
936        if let Err(e) = crate::handlers::config_sync::auto_apply_claude_md(&filter.config_sync) {
937            log::debug!("Failed to auto-apply CLAUDE.md: {}", e);
938        }
939    }
940
941    Ok(())
942}
