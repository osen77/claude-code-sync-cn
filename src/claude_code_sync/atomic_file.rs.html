15impl FileLock {
16    pub(crate) fn acquire(lock_path: &Path) -> Result<Self> {
17        reject_lock_symlink_if_present(lock_path)?;
18        let parent = lock_path.parent().context("lock path has no parent")?;
19        std::fs::create_dir_all(parent)?;
20        let mut options = OpenOptions::new();
21        options.create(true).read(true).write(true);
22        #[cfg(unix)]
23        {
24            use std::os::unix::fs::OpenOptionsExt;
25            options.custom_flags(libc::O_NOFOLLOW);
26            options.mode(0o600);
27        }
28        let file = options.open(lock_path)?;
29        validate_open_lock_path(lock_path)?;
30        set_private_file_permissions(&file)?;
31        FileExt::lock(&file)?;
32        Ok(Self { file })
33    }
34}
35
36impl Drop for FileLock {
37    fn drop(&mut self) {
38        let _ = FileExt::unlock(&self.file);
39    }
40}

42fn reject_lock_symlink_if_present(lock_path: &Path) -> Result<()> {
43    match std::fs::symlink_metadata(lock_path) {
44        Ok(metadata) if metadata.file_type().is_symlink() => {
45            bail!("lock path must not be a symlink: {}", lock_path.display())
46        }
47        Ok(_) => Ok(()),
48        Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(()),
49        Err(error) => Err(error)
50            .with_context(|| format!("failed to inspect lock path: {}", lock_path.display())),
51    }
52}
53
54fn validate_open_lock_path(lock_path: &Path) -> Result<()> {
55    let metadata = std::fs::symlink_metadata(lock_path).with_context(|| {
56        format!(
57            "failed to inspect opened lock path: {}",
58            lock_path.display()
59        )
60    })?;
61    if metadata.file_type().is_symlink() {
62        bail!(
63            "lock path became a symlink while opening: {}",
64            lock_path.display()
65        )
66    }
67    Ok(())
68}

70pub(crate) fn persist_json_atomic<T: Serialize>(target: &Path, value: &T) -> Result<()> {
71    #[cfg(test)]
72    {
73        let call = PERSIST_CALLS.with(|count| {
74            let next = count.get().saturating_add(1);
75            count.set(next);
76            next
77        });
78        if FAIL_ON_PERSIST_CALL.with(|value| value.get() == Some(call)) {
79            anyhow::bail!("test atomic persist failure");
80        }
81    }
82    let parent = target.parent().context("JSON target has no parent")?;
83    std::fs::create_dir_all(parent)?;
84    let bytes = serde_json::to_vec(value)?;
85    let mut temp = NamedTempFile::new_in(parent)?;
86    set_private_file_permissions(temp.as_file())?;
87    temp.write_all(&bytes)?;
88    temp.flush()?;
89    temp.as_file().sync_all()?;
90    temp.persist(target).map_err(|error| error.error)?;
91    sync_parent_directory(parent)?;
92    Ok(())
93}

95/// Synchronize a directory after an atomic child replacement or removal.
96///
97/// Unix supports directory fsync through a read-only directory handle. Windows
98/// and other platforms keep the operation explicit but use the platform's
99/// available atomic rename semantics; the helper remains a testable boundary.
100pub(crate) fn sync_parent_directory(parent: &Path) -> Result<()> {
101    #[cfg(test)]
102    if FORCE_PARENT_SYNC_FAILURE.with(Cell::get) {
103        anyhow::bail!("test parent directory sync failure");
104    }
105
106    #[cfg(unix)]
107    {
108        File::open(parent)
109            .with_context(|| format!("failed to open directory for sync: {}", parent.display()))?
110            .sync_all()
111            .with_context(|| format!("failed to sync directory: {}", parent.display()))?;
112    }
113    #[cfg(windows)]
114    {
115        let _ = parent;
116        // Rust's portable std API does not expose a directory fsync contract on
117        // Windows; callers still receive errors from all file operations.
118    }
119    #[cfg(not(any(unix, windows)))]
120    {
121        let _ = parent;
122    }
123    Ok(())
124}
